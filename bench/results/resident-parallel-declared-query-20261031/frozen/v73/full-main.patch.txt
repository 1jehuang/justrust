+/// Native final graph with the original wave ordering and parallel cold lowering.
+#[salsa::tracked(returns(ref))]
+fn parallel_crate_declarations(
+    db: &dyn DeclarationDatabase,
+    krate: ide_db::base_db::Crate,
+) -> Declared {
+    #[cfg(test)]
+    PARALLEL_DECLARED_QUERY_PROVIDERS.set(PARALLEL_DECLARED_QUERY_PROVIDERS.get() + 1);
+    // A process-once witness of provider entry only. Failed stderr writes are ignored.
+    static FIRST_PROVIDER: std::sync::Once = std::sync::Once::new();
+    FIRST_PROVIDER.call_once(|| {
+        use std::io::Write;
+        let _ = writeln!(
+            std::io::stderr().lock(),
+            "parallel-declared-query-provider-entry pid={}",
+            std::process::id()
+        );
+    });
+    hir_def::nameres::crate_def_map(db, krate);
+    let mut declared = module_declarations(db, krate).clone();
+    while !declared.bodies.is_empty() || !declared.provided.is_empty() {
+        for under_provided in [false, true] {
+            let bodies = if under_provided {
+                std::mem::take(&mut declared.provided)
+            } else {
+                std::mem::take(&mut declared.bodies)
+            };
+            // Every fragment is read on the parent after warmup. Child snapshot
+            // reads alone never establish this tracked parent's dependencies.
+            for body in prewarm_declaration_bodies(db, bodies) {
+                declared.extend(body_declarations(db, body, under_provided));
+            }
+        }
+    }
+    declared
+}

+/// Keep small waves on the already attached parent. The original lower's small
+/// branch attaches its argument, so a cloned database there would violate HIR's
+/// attached-database identity invariant even though its Salsa storage is shared.
+fn prewarm_declaration_bodies(
+    db: &dyn DeclarationDatabase,
+    bodies: Vec<DefWithBodyId>,
+) -> Vec<DefWithBodyId> {
+    // Matches the unchanged original lower's PER_THREAD threshold. Empty and
+    // single-worker waves must not clone or reattach the database on this thread.
+    let parallel =
+        bodies.len() > 16 && std::thread::available_parallelism().is_ok_and(|it| it.get() > 1);
+    if parallel {
+        #[cfg(test)]
+        DECLARATION_PREWARM_ROUTES.with(|routes| {
+            let (parent, parallel) = routes.get();
+            routes.set((parent, parallel + 1));
+        });
+        lower(&db.declaration_snapshot(), bodies)
+    } else {
+        #[cfg(test)]
+        DECLARATION_PREWARM_ROUTES.with(|routes| {
+            let (parent, parallel) = routes.get();
+            routes.set((parent + 1, parallel));
+        });
+        for &body in &bodies {
+            Body::of(db, body);
+        }
+        bodies
+    }
+}

+    fn parallel_declared_query_full_walk(db: &RootDatabase, krate: Crate) -> Declared {
+        let mut expected = Declared::default();
+        expected.in_modules(db, krate);
+        while !expected.bodies.is_empty() || !expected.provided.is_empty() {
+            for under_provided in [false, true] {
+                let bodies = if under_provided {
+                    std::mem::take(&mut expected.provided)
+                } else {
+                    std::mem::take(&mut expected.bodies)

+                    let mut fragment = Declared { under_provided, ..Declared::default() };
+                    for (_, def_map) in Body::of(db, body).blocks(db) {
+                        for (_, module) in def_map.modules(db) {
+                            fragment.in_scope(db, module.scope(db));
+                        }
+                    }
+                    expected.extend(&fragment);
+                }
+            }
+        }
